Unggah kode C++ (Arduino IDE) di bawah ini ke ESP8266/ESP32 Anda. Sistem ini menggunakan Firebase API (REST) yang aman dan terstruktur untuk mengirim data langsung ke dashboard ini. 

/* 
 * Membutuhkan Library: "FirebaseClient" oleh mobizt
 * Board: ESP32 atau ESP8266
 */
#include <WiFi.h> // Gunakan ESP8266WiFi.h untuk ESP8266
#include <FirebaseClient.h>

const char* WIFI_SSID = "NAMA_WIFI_ANDA";
const char* WIFI_PASSWORD = "PASSWORD_WIFI";

// Konfigurasi akan menyesuaikan dengan project Anda saat di-deploy
#define PROJECT_ID "ID_PROJECT_FIREBASE" 

void setup() {
  Serial.begin(115200);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500); Serial.print(".");
  }
  Serial.println("\nTerhubung ke WiFi!");
  
  // Inisialisasi pin Sensor pH Analog disini...
}

void loop() {
  // Contoh pembacaan sensor (Ganti dengan analogRead sensor Anda)

}
